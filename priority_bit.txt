classes = {
    "0" : ("Class A", "0.0.0.0", "127.255.255.255", "255.0.0.0"),
    "10" : ("Class B", "128.0.0.0", "191.255.255.255", "255.255.0.0"),
    "110" : ("Class C", "192.0.0.0", "223.255.255.255", "255.255.255.0"),
    "1110" : ("Class D", "224.0.0.0", "251.255.255.255", "N/A"),
    "1111" : ("Class E", "252.0.0.0", "255.255.255.255", "N/A")
}

bits = input("Enter the Digit of the Priority 0/10/110/1110/1111")

if bits in classes :
  name, start, end, mask = classes[bits]
  print("Class Type", name)
  print("Priority Bits", bits)
  print("Start IP Address", start)
  print("End IP Address", end)
  print("Subnet Mask", mask)
else :
  print("Invalid Priority Bits")
